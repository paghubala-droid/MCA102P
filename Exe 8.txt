DECLARE
acc NUMBER := &acc_no;
amt NUMBER := &withdraw_amt;
bal NUMBER;
BEGIN
SELECT Balance INTO bal FROM Bank_Account WHERE Account_No = acc;
IF amt > bal THEN
DBMS_OUTPUT.PUT_LINE('Insufficient Balance! Available = ' ||
bal);
ELSE
UPDATE Bank_Account SET Balance = Balance - amt WHERE Account_No
= acc;
DBMS_OUTPUT.PUT_LINE('Withdrawal Successful. New Balance =
' || (bal-amt));
COMMIT;
END IF;
EXCEPTION
WHEN NO_DATA_FOUND THEN
DBMS_OUTPUT.PUT_LINE('Invalid Account Number!');
END;
/
